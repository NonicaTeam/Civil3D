13    };
14
15    foreach (var corridorId in _civildoc.CorridorCollection)

18        ObjectIdCollection solids = corridor.ExportSolids(exportParams, _database);
19        write($"\nExported {solids.Count} solids or bodies in {corridor.Name}.");
20    }
