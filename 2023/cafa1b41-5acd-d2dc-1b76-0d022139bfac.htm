 3 ObjectId pointGroupId = _civildoc.PointGroups.Add("Example Point Group");

 5
 6 StandardPointGroupQuery standardQuery = new StandardPointGroupQuery();
 7
 8 // Include properties:

13 standardQuery.IncludeRawDescriptions = "BRK*";
14

21
